## 修复前（main @ 17a21e4）

    SELECT COUNT(DISTINCT b.id) AS bookings, COUNT(rn.id) AS room_nights
    FROM bookings b JOIN rooms r ON r.id = b.room_id
    LEFT JOIN room_nights rn ON rn.booking_id = b.id
    WHERE b.room_type_id <> r.room_type_id;

    | bookings | room_nights |

    |        7 |          13 |

    SELECT name, COUNT(*) AS types, GROUP_CONCAT(code ORDER BY id) AS codes
    FROM room_types GROUP BY name HAVING COUNT(*) > 1;
